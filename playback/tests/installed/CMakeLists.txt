cmake_minimum_required(VERSION 3.20)
project(SaltsCorePlaybackInstalled LANGUAGES C CXX)
include(CTest)
find_package(SaltsCore 0.1 CONFIG REQUIRED)
if(NOT TARGET Salts::Playback)
  message(FATAL_ERROR "Playback qualification requires installed Salts::Playback")
endif()
get_target_property(_kind Salts::Playback TYPE)
get_target_property(_imported Salts::Playback IMPORTED)
if(NOT _imported OR NOT _kind STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "Playback must have one installed shared native owner")
endif()
add_executable(playback_consumer_c playback_consumer.c)
target_compile_features(playback_consumer_c PRIVATE c_std_11)
target_link_libraries(playback_consumer_c PRIVATE Salts::Playback)
add_test(NAME sdk.playback_c11 COMMAND playback_consumer_c)
add_executable(playback_consumer_cpp playback_consumer.cpp)
target_compile_features(playback_consumer_cpp PRIVATE cxx_std_17)
target_link_libraries(playback_consumer_cpp PRIVATE Salts::Playback)
add_test(NAME sdk.playback_cpp17 COMMAND playback_consumer_cpp)
foreach(_target IN ITEMS playback_consumer_c playback_consumer_cpp)
  if(MSVC)
    target_compile_options(${_target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${_target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
