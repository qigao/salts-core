cmake_minimum_required(VERSION 3.21)
project(SaltsCryptoInstalledOwner LANGUAGES C CXX)
include(CTest)
find_package(Crypto 0.1 CONFIG REQUIRED)
if(NOT TARGET Salts::ASN1)
  message(FATAL_ERROR "Installed Crypto must import standalone Salts::ASN1")
endif()
get_target_property(_asn1_type Salts::ASN1 TYPE)
if(NOT _asn1_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Salts::ASN1 must remain STATIC")
endif()
if(NOT TARGET Salts::Crypto)
  message(FATAL_ERROR "Installed Salts::Crypto missing")
endif()
get_target_property(_kind Salts::Crypto TYPE)
get_target_property(_installed Salts::Crypto IMPORTED)
if(NOT _kind STREQUAL "STATIC_LIBRARY" OR NOT _installed)
  message(FATAL_ERROR "Exactly one imported STATIC Salts::Crypto is required")
endif()
foreach(_split IN ITEMS Salts::TLS Salts::TLSHost Salts::TLSCipherBIO Salts::CryptoProvider Crypto::ASN1)
  if(TARGET "${_split}")
    message(FATAL_ERROR "Separate native Crypto target forbidden: ${_split}")
  endif()
endforeach()
add_executable(crypto_owner_c11 consumer.c)
target_compile_features(crypto_owner_c11 PRIVATE c_std_11)
target_link_libraries(crypto_owner_c11 PRIVATE Salts::Crypto)
add_test(NAME crypto.static.installed_c11_sm3_tls COMMAND crypto_owner_c11)
add_executable(crypto_owner_cpp17 consumer.cpp)
target_compile_features(crypto_owner_cpp17 PRIVATE cxx_std_17)
target_link_libraries(crypto_owner_cpp17 PRIVATE Salts::Crypto)
add_test(NAME crypto.static.installed_cpp17_sm3_tls COMMAND crypto_owner_cpp17)
