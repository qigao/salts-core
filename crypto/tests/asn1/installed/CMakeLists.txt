cmake_minimum_required(VERSION 3.21)
project(CryptoASN1InstalledConsumer LANGUAGES C CXX)
include(CTest)
find_package(Crypto 0.1 CONFIG REQUIRED)
if(NOT TARGET Salts::Crypto)
  message(FATAL_ERROR "ASN.1 requires the installed Salts::Crypto owner")
endif()
if(NOT TARGET Salts::ASN1)
  message(FATAL_ERROR "Crypto must import canonical Salts::ASN1")
endif()
get_target_property(_asn1_kind Salts::ASN1 TYPE)
if(NOT _asn1_kind STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Salts::ASN1 must be a separate static library")
endif()
if(TARGET SaltsCore::ASN1 OR TARGET Crypto::ASN1)
  message(FATAL_ERROR "Separate/legacy ASN.1 target forbidden")
endif()
get_target_property(_kind Salts::Crypto TYPE)
get_target_property(_imported Salts::Crypto IMPORTED)
if(NOT _imported OR NOT _kind STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "ASN.1 must be linked from installed static Salts::Crypto")
endif()
add_executable(crypto_asn1_installed_c11 consumer.c)
target_compile_features(crypto_asn1_installed_c11 PRIVATE c_std_11)
target_link_libraries(crypto_asn1_installed_c11 PRIVATE Salts::Crypto)
add_test(NAME crypto.asn1.installed_c11 COMMAND crypto_asn1_installed_c11)
add_executable(crypto_asn1_installed_cpp17 consumer.cpp)
target_compile_features(crypto_asn1_installed_cpp17 PRIVATE cxx_std_17)
target_link_libraries(crypto_asn1_installed_cpp17 PRIVATE Salts::Crypto)
add_test(NAME crypto.asn1.installed_cpp17 COMMAND crypto_asn1_installed_cpp17)
