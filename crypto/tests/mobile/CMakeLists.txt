cmake_minimum_required(VERSION 3.21)
project(CryptoMobileInstalledSmoke LANGUAGES C CXX)
if(NOT DEFINED CRYPTO_SDK_ROOT OR NOT IS_DIRECTORY "${CRYPTO_SDK_ROOT}")
  message(FATAL_ERROR "CRYPTO_SDK_ROOT requires the exact installed target SDK")
endif()
find_package(Crypto 0.1 CONFIG REQUIRED
  PATHS "${CRYPTO_SDK_ROOT}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
if(NOT TARGET Salts::Crypto)
  message(FATAL_ERROR "Canonical static Crypto target missing")
endif()
get_target_property(_kind Salts::Crypto TYPE)
get_target_property(_imported Salts::Crypto IMPORTED)
get_target_property(_archive Salts::Crypto IMPORTED_LOCATION_RELEASE)
if(NOT _kind STREQUAL "STATIC_LIBRARY" OR NOT _imported OR
   NOT EXISTS "${_archive}")
  message(FATAL_ERROR "Installed Crypto must import one valid STATIC archive")
endif()
foreach(_split IN ITEMS Salts::TLS Salts::CryptoProvider Crypto::Crypto)
  if(TARGET ${_split})
    message(FATAL_ERROR "Unacceptable split/alias target ${_split}")
  endif()
endforeach()

# iOS cross compilation builds actual consumer objects without device runtime.
# Android additionally links the installed native archive into executables.
foreach(_lang IN ITEMS c11 cpp17)
  if(_lang STREQUAL c11)
    set(_source "${CMAKE_CURRENT_SOURCE_DIR}/../owner/consumer.c")
    set(_feature c_std_11)
  else()
    set(_source "${CMAKE_CURRENT_SOURCE_DIR}/../owner/consumer.cpp")
    set(_feature cxx_std_17)
  endif()
  add_library(crypto_mobile_${_lang}_headers OBJECT "${_source}")
  target_compile_features(crypto_mobile_${_lang}_headers PRIVATE ${_feature})
  target_link_libraries(crypto_mobile_${_lang}_headers PRIVATE Salts::Crypto)
  if(CRYPTO_MOBILE_EXE_LINK_TESTS)
    add_executable(crypto_mobile_${_lang}_linked "${_source}")
    target_compile_features(crypto_mobile_${_lang}_linked PRIVATE ${_feature})
    target_link_libraries(crypto_mobile_${_lang}_linked PRIVATE Salts::Crypto)
  endif()
endforeach()
