cmake_minimum_required(VERSION 3.21)
project(ASN1CryptoFormatSourceQualification LANGUAGES C CXX)
include(CTest)
if(NOT DEFINED CRYPTO_SOURCE OR NOT IS_DIRECTORY "${CRYPTO_SOURCE}")
  message(FATAL_ERROR "Require explicit exact Crypto source directory")
endif()
find_package(ASN1 0.1 CONFIG REQUIRED)
if(NOT TARGET Salts::ASN1)
  message(FATAL_ERROR "Crypto key-format tests require canonical Salts::ASN1")
endif()
if(TARGET GmSSL::GmSSL OR TARGET Salts::Crypto)
  message(FATAL_ERROR "Format source qualification must not link GmSSL/Crypto")
endif()

add_library(crypto_keyformat_objects OBJECT "${CRYPTO_SOURCE}/src/asn1/crypto.c")
set_target_properties(crypto_keyformat_objects PROPERTIES
  C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(crypto_keyformat_objects PRIVATE "${CRYPTO_SOURCE}/include")
target_link_libraries(crypto_keyformat_objects PUBLIC Salts::ASN1)

foreach(_test IN ITEMS crypto header_cpp)
  if(_test STREQUAL "crypto")
    set(_src "${CRYPTO_SOURCE}/tests/asn1/crypto.c")
    set(_name asn1.crypto_formats_c11)
  else()
    set(_src "${CRYPTO_SOURCE}/tests/asn1/header_cpp.cpp")
    set(_name asn1.crypto_formats_cpp17)
  endif()
  add_executable(format_${_test} "${_src}"
    $<TARGET_OBJECTS:crypto_keyformat_objects>)
  target_include_directories(format_${_test} PRIVATE
    "${CRYPTO_SOURCE}/include"
    "${CRYPTO_SOURCE}/tests/asn1")
  target_link_libraries(format_${_test} PRIVATE Salts::ASN1)
  if(_test STREQUAL "crypto")
    target_compile_features(format_${_test} PRIVATE c_std_11)
  else()
    target_compile_features(format_${_test} PRIVATE cxx_std_17)
  endif()
  add_test(NAME ${_name} COMMAND format_${_test})
endforeach()
