cmake_minimum_required(VERSION 3.21)
project(ASN1InstalledConsumer LANGUAGES C CXX)
include(CTest)
find_package(ASN1 0.1 CONFIG REQUIRED)
if(NOT TARGET Salts::ASN1)
  message(FATAL_ERROR "Missing installed Salts::ASN1")
endif()
get_target_property(_kind Salts::ASN1 TYPE)
get_target_property(_imported Salts::ASN1 IMPORTED)
if(NOT _imported OR NOT _kind STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Salts::ASN1 must be imported STATIC")
endif()
foreach(_forbidden IN ITEMS Salts::Crypto Salts::CMeta Salts::Core
  GmSSL::GmSSL Crypto::ASN1 SaltsNet::Asn1Parser)
  if(TARGET "${_forbidden}")
    message(FATAL_ERROR "Standalone ASN.1 unexpectedly brought in ${_forbidden}")
  endif()
endforeach()
foreach(_lang IN ITEMS c cpp)
  add_executable(asn1_installed_${_lang} consumer.${_lang})
  if(_lang STREQUAL "c")
    target_compile_features(asn1_installed_${_lang} PRIVATE c_std_11)
  else()
    target_compile_features(asn1_installed_${_lang} PRIVATE cxx_std_17)
  endif()
  target_link_libraries(asn1_installed_${_lang} PRIVATE Salts::ASN1)
  add_test(NAME asn1.installed_${_lang} COMMAND asn1_installed_${_lang})
endforeach()
